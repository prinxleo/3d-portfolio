function doGet() {
  var cache = CacheService.getScriptCache();
  var cached = cache.get("portfolio_sync_v1");
  if (cached) return ContentService.createTextOutput(cached).setMimeType(ContentService.MimeType.JSON);

  var folderIds = ["1Vb_05qyweeF3cIrs0HRfMZdHGY8H-Lwx", "1kIXHapwbXL8_03OdnDXHt2epKjFRu9Jm"];
  var out = [];
  folderIds.forEach(function(rootId, colIdx) {
    var root = DriveApp.getFolderById(rootId);
    var subs = root.getFolders();
    while (subs.hasNext()) {
      var sf = subs.next();
      var files = sf.getFiles();
      var imgs = [], desc = "";
      while (files.hasNext()) {
        var f = files.next();
        var mime = f.getMimeType() || "";
        var name = f.getName() || "";
        // Strictly ignore 3D source files (.fbx, .obj, .blend, .glb, .zip, .rar)
        if (mime.indexOf("image/") === 0 && !/(_normal|_roughness|_metallic|_ao|_height)/i.test(name)) {
          imgs.push({ id: f.getId(), t: "Preview Angle " + (imgs.length + 1) });
        } else if (name.toLowerCase().indexOf(".txt") !== -1 && !desc) {
          try { desc = f.getBlob().getDataAsString().slice(0, 500); } catch(e) {}
        }
      }
      if (imgs.length > 0) {
        out.push({
          sub: sf.getName(),
          title: sf.getName(),
          col: colIdx === 0 ? "Collection 1" : "Collection 2",
          date: Utilities.formatDate(sf.getLastUpdated(), "UTC", "yyyy-MM-dd"),
          desc: desc,
          imgs: imgs.slice(0, 8)
        });
      }
    }
  });
  var json = JSON.stringify({ status: "ok", models: out });
  try { cache.put("portfolio_sync_v1", json, 600); } catch(e) {}
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
